import numpy as np
from sklearn.cluster import KMeans
data = np.array([
    [10, 2, 1],
    [15, 3, 2],
    [20, 4, 3],
    [60, 15, 10],
    [70, 18, 12],
    [80, 20, 15],
    [30, 5, 4],
    [35, 7, 5]
])
model = KMeans(n_clusters=3, random_state=42, n_init=10)
model.fit(data)
def distance(x, c):
    return np.sqrt(np.sum((x - c) ** 2))
for i, x in enumerate(data):
    d = [distance(x, c) for c in model.cluster_centers_]
    print("User", i + 1, "-> Group", np.argmin(d) + 1)
