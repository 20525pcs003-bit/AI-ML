P_selected = 0.6
P_rejected = 0.4

P_skill_selected = 0.9
P_skill_rejected = 0.3

skill = input("Has good programming skills? (yes/no): ")

if skill.lower() == "yes":

    P_skill = (P_skill_selected * P_selected) + (P_skill_rejected * P_rejected)

    selected = (P_skill_selected * P_selected) / P_skill
    rejected = (P_skill_rejected * P_rejected) / P_skill

else:

    selected = 1 - P_skill_selected
    rejected = 1 - P_skill_rejected

print("Selected =", round(selected, 2))
print("Rejected =", round(rejected, 2))

if selected > rejected:
    print("Prediction: SELECTED")
else:
    print("Prediction: REJECTED")
