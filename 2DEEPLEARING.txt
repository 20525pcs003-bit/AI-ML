w1, w2 = 0.5, 0.4
h1, h2 = 0.6, 0.3
b1, b2 = 0.1, 0.1
out_w1, out_w2 = 0.7, 0.5
b3 = 0
def sigmoid(x):
    return 1 / (1 + 2.71828 ** (-x))
income = float(input("Enter Income: "))
credit = float(input("Enter Credit Score: "))
x1 = income / 100000
x2 = credit / 850
z1 = x1 * w1 + x2 * w2 + b1
h1_out = sigmoid(z1)
z2 = h1_out * h1 + x2 * h2 + b2
h2_out = sigmoid(z2)
z3 = h2_out * out_w1 + h1_out * out_w2 + b3
output = sigmoid(z3)
print("Approval Probability:", round(output, 2))
if output >= 0.7:
    print("Prediction: Loan Approved")
else:
    print("Prediction: Loan Not Approved")
