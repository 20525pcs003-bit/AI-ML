import pandas as pd
import matplotlib.pyplot as plt
from sklearn.tree import DecisionTreeClassifier

df = pd.DataFrame({
    "Amount": [100, 250, 500, 50, 900, 20, 700, 60],
    "Time": [1, 2, 3, 4, 5, 6, 7, 8],
    "class": [0, 0, 1, 0, 1, 0, 1, 0]
})

x = df[["Amount", "Time"]]
y = df["class"]

model = DecisionTreeClassifier()
model.fit(x, y)

card = pd.DataFrame(
    [[600, 9]],
    columns=["Amount", "Time"]
)

result = model.predict(card)

if result[0] == 1:
    print("Fraud Transaction")
else:
    print("Normal Transaction")

normal = sum(df["class"] == 0)
fraud = sum(df["class"] == 1)

plt.pie(
    [normal, fraud],
    labels=["Normal", "Fraud"],
    colors=["yellow", "pink"],
    autopct="%1.1f%%"
)

plt.title("Credit card fraud detection")
plt.savefig("fraud_graph.png")
