w1, w2 = 0.6, 0.4
b = 0
def sigmoid(x):
    return 1 / (1 + 2.71828 ** (-x))
distance = float(input("Enter Distance (km): "))
traffic = float(input("Enter Traffic Level (1-10): "))
x1 = distance / 100
x2 = traffic / 10
z = x1 * w1 + x2 * w2 + b
output = sigmoid(z)
print("Prediction:", round(output, 2))
if output >= 0.5:
    print("Delivery Time: Long")
else:
    print("Delivery Time: Short")
