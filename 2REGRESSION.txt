import pandas as pd
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt

d = pd.DataFrame({
    "Area": [1000, 1500, 2000, 2500, 3000],
    "price": [20, 30, 40, 50, 60]
})

m = LinearRegression().fit(d[["Area"]], d["price"])

a = float(input("Enter Area: "))
p = m.predict([[a]])

print("Price: ₹", p[0] * 100000)

plt.scatter(d.Area, d.price, color="blue", label="Actual")
plt.scatter(a, p, color="red", s=100, label="Prediction")

plt.xlabel("Area (sq.ft)")
plt.ylabel("Price (lakhs)")
plt.title("House Price Prediction")
plt.legend()
plt.savefig("house_price_graph.png")
plt.show()
