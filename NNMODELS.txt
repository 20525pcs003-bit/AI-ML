import numpy as np
X = np.array([
    [20, 80, 5],
    [50, 50, 10],
    [80, 20, 15],
    [90, 10, 20],
    [40, 60, 8]
])
y = np.array([0, 0, 1, 1, 0])
W1 = np.array([[0.2, 0.1], [0.3, 0.2], [0.1, 0.4]])
b1 = np.array([0.1, 0.1])
W2 = np.array([0.5, 0.6])
b2 = 0.1
H = np.maximum(0, X @ W1 + b1)
O = 1 / (1 + np.exp(-(H @ W2 + b2)))
prediction = (O >= 0.5).astype(int)
print("Congestion Predictions:", prediction)
new_data = np.array([[75, 25, 18]])
H = np.maximum(0, new_data @ W1 + b1)
O = 1 / (1 + np.exp(-(H @ W2 + b2)))
print("Congestion Probability:", round(O[0], 2))
print("Prediction:", "HIGH CONGESTION" if O[0] >= 0.5 else "LOW CONGESTION")
