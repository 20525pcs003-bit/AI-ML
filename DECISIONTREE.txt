import pandas as pd
import matplotlib.pyplot as plt
from sklearn.tree import DecisionTreeClassifier
df=pd.DataFrame({
"Fever":[1,1,0,1,0],
"Cough":[1,0,1,1,0],
"Disease":[1,1,0,1,0]})
X=df[["Fever","Cough"]]
y=df["Disease"]
model=DecisionTreeClassifier().fit(X,y)

p=model.predict(pd.DataFrame([[1,1]],columns=X.columns))[0]
print("Diagnosis:", "Positive" if p else "Negative")

v=y.value_counts().sort_index()
plt.pie(v,labels=["Negative","Positive"],autopct="%.1f%%",
        startangle=90,wedgeprops={"width":.4})
plt.title("Medical Diagnosis")
plt.savefig("output.png")
plt.show()
