from sklearn.ensemble import RandomForestClassifier
import matplotlib.pyplot as plt
X = [[30, 1], [40, 1], [50, 2], [60, 2], [70, 3], [80, 3]]
y = [0, 0, 1, 1, 2, 2]
model = RandomForestClassifier(n_estimators=10, random_state=42)
model.fit(X, y)
p = model.predict([[85, 8]])
print("Traffic accident severity prediction")
print("-----------------------------------")
print("Model: Random Forest")
print("Predicted severity:", ["MINOE","SERIOUS","FATAL"][p[0]])
speed = [30, 40, 50, 60, 70, 80]
plt.scatter(speed, y)
plt.title("Traffic accident severity prediction")
plt.xlabel("Speed")
plt.ylabel("Severity")
plt.savefig("output.png")
plt.show()
