P_engine = 0.25
P_noise_given_engine = 0.85
P_noise = 0.40

P_engine_given_noise = (P_noise_given_engine * P_engine) / P_noise

print("Car Engine Analysis")
print("----------")
print("P(Engine fault) =", P_engine)
print("P(Noise | Engine fault) =", P_noise_given_engine)
print("P(Noise) =", P_noise)
print("Probability of Engine Fault:")
print(round(P_engine_given_noise, 2))
