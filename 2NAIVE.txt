play = 6 / 10
not_play = 4 / 10

P_sunny_play = 2 / 6
P_sunny_not = 3 / 4

P_high_play = 3 / 4 * 2 / 6
P_high_not = 4 / 4

play_prob = play * P_sunny_play * P_high_play
not_prob = not_play * P_sunny_not * P_high_not

if play_prob > not_prob:
    print("Prediction : play cricket")
else:
    print("Prediction : Don't play cricket")
