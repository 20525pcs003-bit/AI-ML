import numpy as np
np.random.seed(42)
X = np.array([[230,1.2,35,2],
              [220,1,40,3],
              [210,0.8,50,5],
              [200,0.6,55,7],
              [240,1.3,30,1],
              [205,0.7,60,8]])
Y = np.array([0,0,1,1,0,1])
max_vals = X.max(axis=0)
X = X / max_vals
W = np.random.randn(4)
b = 0
sigmoid = lambda z: 1/(1+np.exp(-z))
for _ in range(1000):
    z = X @ W + b
    P = sigmoid(z)
    error = Y - P
    W += 0.1 * (X.T @ error)
    b += 0.1 * error.sum()
new_light = np.array([195,0.5,58,6]) / max_vals
probability = sigmoid(new_light @ W + b)
print("Failure Probability:", round(float(probability),2))
print("Prediction:", "FAILURE" if probability >= 0.5 else "NORMAL")
