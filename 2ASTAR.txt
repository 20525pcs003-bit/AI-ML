import random
class Student:
    def __init__(self, marks):
        self.marks = marks
    def __repr__(self):
        return "Student with %d marks." % self.marks
students = [Student(random.randint(35, 100)) for _ in range(5)]
print("\nRandom students:")
print(students)
def student_marks(student):
    return student.marks
print("Marks of first student:")
print(student_marks(students[0]))
print()
top_student = max(students, key=student_marks)
print("Top student:", top_student)
